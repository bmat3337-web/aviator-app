    <script>
        // Initialize Telegram Web App API
        const tg = window.Telegram.WebApp;
        tg.expand();

        let sessionMultipliers = [];
        let currentStreak = 0;

        function calculateWagers() {
            const br = parseFloat(document.getElementById('bankroll').value) || 0;
            const totalRisk = br * 0.01; // Strict 1% risk rule
            
            // Apportion via 2.5 : 1 scale strategy framework
            const bet2 = totalRisk / 3.5;
            const bet1 = bet2 * 2.5;

            document.getElementById('bet1').innerText = `$${bet1.toFixed(2)}`;
            document.getElementById('bet2').innerText = `$${bet2.toFixed(2)}`;
        }

        function submitValue() {
            const input = document.getElementById('multiplierInput');
            const val = parseFloat(input.value);
            if (isNaN(val) || val < 1.0) return;

            sessionMultipliers.unshift(val); // Add to front of history log
            
            // Calculate streaks
            if (val < 2.00) {
                currentStreak++;
            </div> else {
                currentStreak = 0;
            }

            // Calculate Trap Distribution
            const traps = sessionMultipliers.filter(m => m < 2.00).length;
            const trapPct = (traps / sessionMultipliers.length) * 100;

            // Update DOM Elements
            document.getElementById('currentStreak').innerText = currentStreak;
            document.getElementById('trapRate').innerText = `${trapPct.toFixed(1)}%`;
            
            // Append to screen log view
            const logBox = document.getElementById('historyLog');
            logBox.innerHTML = sessionMultipliers.map(m => `${m.toFixed(2)}x`).join(' | ');

            // Reset input field
            input.value = '';
            calculateWagers();

            // Send analytical data packet back to Telegram Bot backend if required
            tg.sendData(JSON.stringify({multiplier: val, currentStreak: currentStreak}));
        }

        // Run initial configuration setup
        calculateWagers();
    </script>
